20// FlowPlus extension inside the inherited DDRG/DFTR frames. The server supplies
21// the source identity from the authenticated connection, never from file metadata.
22struct RoutedPayload
23{
24 QString source;
25 QString target;

27};

29inline QByteArray encode(const QString &source, const QString &target, const QByteArray &data)
30{
31 if (source.toUtf8().size() > 255 || target.toUtf8().size() > 255 || data.size() > kMaxPayload ||
32 source.contains(QChar::Null) || target.contains(QChar::Null)) {
33 return {};
34 }
35 const auto header = QJsonDocument(
36 QJsonObject{
37 {QStringLiteral("flowplusRoute"), 1},
38 {QStringLiteral("source"), source},
39 {QStringLiteral("target"), target}
40 }
41 ).toJson(QJsonDocument::Compact);
42 if (header.size() > kMaxRouteHeader) {
43 return {};
44 }
45 return header + '\n' + data;
46}
47
48inline bool decode(const QByteArray &wire, RoutedPayload &result)
49{
50 const auto separator = wire.indexOf('\n');
51 if (separator <= 0 || separator > kMaxRouteHeader || wire.size() > kMaxPayload + kMaxRouteHeader + 1) {
52 return false;
53 }
54 QJsonParseError error;
55 const auto document = QJsonDocument::fromJson(wire.first(separator), &error);
56 if (error.error != QJsonParseError::NoError || !document.isObject()) {
57 return false;
58 }
59 const auto header = document.object();
60 if (header.value(QStringLiteral("flowplusRoute")).toInt() != 1 ||
61 !header.value(QStringLiteral("source")).isString() || !header.value(QStringLiteral("target")).isString()) {
62 return false;
63 }
64 result = {
65 header.value(QStringLiteral("source")).toString(), header.value(QStringLiteral("target")).toString(),
66 wire.mid(separator + 1)
67 };
68 return result.source.toUtf8().size() <= 255 && result.target.toUtf8().size() <= 255 &&
69 result.data.size() <= kMaxPayload && !result.source.contains(QChar::Null) &&
70 !result.target.contains(QChar::Null);
71}

73struct NumberedPayload
74{
75 quint32 number = 0;
76 QByteArray wire;
77 RoutedPayload route;
78};

80inline bool decodeNumberedIpc(const QString &arguments, quint32 maximum, NumberedPayload &result)
81{
82 const auto separator = arguments.indexOf(':');
83 if (separator <= 0 || arguments.size() > kMaxIpcMessage) {
84 return false;
85 }
86 bool validNumber = false;
87 result.number = arguments.first(separator).toUInt(&validNumber);
88 if (!validNumber || result.number > maximum) {
89 return false;
90 }
91 result.wire =
92 QByteArray::fromBase64(arguments.mid(separator + 1).toLatin1(), QByteArray::AbortOnBase64DecodingErrors);
93 return decode(result.wire, result.route);
94}

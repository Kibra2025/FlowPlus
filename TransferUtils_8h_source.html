22inline bool isPortableFileName(const QString &name)
23{
24 if (name.isEmpty() || name.size() > kMaxFileNameLength || name.toUtf8().size() > kMaxFileNameLength ||
25 name == QStringLiteral(".") || name == QStringLiteral("..") || name.contains('/') || name.contains('\\')) {
26 return false;
27 }
28 for (const auto character : name) {
29 if (character.unicode() < 32 || character.unicode() == 127 || QStringLiteral(":*?\"<>|").contains(character)) {
30 return false;
31 }
32 }
33 const auto stem = name.section('.', 0, 0).trimmed().toUpper();
34 const bool reservedPort = stem.size() == 4 && (stem.startsWith("COM") || stem.startsWith("LPT")) &&
35 ((stem.at(3) >= QChar('1') && stem.at(3) <= QChar('9')) ||
36 QStringLiteral("\u00b9\u00b2\u00b3").contains(stem.at(3)));
37 return !name.endsWith('.') && !name.endsWith(' ') && stem != QStringLiteral("CON") && stem != QStringLiteral("PRN") &&
38 stem != QStringLiteral("AUX") && stem != QStringLiteral("NUL") && !reservedPort;
39}

50inline bool validateTransferManifest(const QJsonArray &manifest, qint64 &totalBytes)
51{
52 totalBytes = 0;
53 if (manifest.isEmpty() || manifest.size() > kMaxTransferFiles) {
54 return false;
55 }
56 for (const auto &entry : manifest) {
57 if (!entry.isObject()) {
58 return false;
59 }
60 const auto object = entry.toObject();
61 const auto sizeValue = object.value(QStringLiteral("size"));
62 const auto size = sizeValue.toDouble(-1);
63 if (!isPortableFileName(object.value(QStringLiteral("fileName")).toString()) || !sizeValue.isDouble() ||
64 !std::isfinite(size) || size < 0 || std::floor(size) != size || size > kMaxTransferBytes - totalBytes) {
65 return false;
66 }
67 totalBytes += static_cast<qint64>(size);
68 }
69 return true;
70}

72inline QString uniqueTransferDestination(const QString &directory, const QString &fileName)
73{
74 if (!isPortableFileName(fileName)) {
75 return {};
76 }
77 const QFileInfo info(fileName);
78 for (int counter = 0; counter < 10000; ++counter) {
79 auto candidateName = fileName;
80 if (counter > 0) {
81 auto base = info.suffix().isEmpty() ? fileName : info.completeBaseName();
82 auto suffix = info.suffix().isEmpty() ? QString() : QStringLiteral(".") + info.suffix();
83 const auto counterText = QStringLiteral(" (%1)").arg(counter);
84 if ((counterText + suffix).toUtf8().size() >= kMaxFileNameLength) {
85 base = fileName;
86 suffix.clear();
87 }
88 const auto tail = counterText + suffix;
89 while (!base.isEmpty() &&
90 ((base + tail).size() > kMaxFileNameLength || (base + tail).toUtf8().size() > kMaxFileNameLength)) {
91 base.chop(1);
92 if (!base.isEmpty() && base.back().isHighSurrogate()) {
93 base.chop(1);
94 }
95 }
96 candidateName = base + tail;
97 if (!isPortableFileName(candidateName)) {
98 return {};
99 }
100 }
101 const auto candidate = QDir(directory).filePath(candidateName);
102 const QFileInfo candidateInfo(candidate);
103 // A dangling symlink is also occupied. Never follow or replace it.
104 if (!candidateInfo.exists() && !candidateInfo.isSymLink()) {
105 return candidate;
106 }
107 }
108 return {};
109}

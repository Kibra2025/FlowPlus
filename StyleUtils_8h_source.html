88inline bool isDarkMode()
89{
90 const QPalette defaultPalette;
91 const auto text = defaultPalette.color(QPalette::WindowText);
92 const auto window = defaultPalette.color(QPalette::Window);
93 return text.lightness() > window.lightness();
94}
99inline QString iconMode()
100{

104inline void updateIconTheme()
105{
106 // Sets the fallback icon path and fallback theme
107 const auto themeName = QStringLiteral("%1-%2").arg(kIconThemeId, iconMode());
108 if (QIcon::themeName().isEmpty() || QIcon::themeName().startsWith(kIconThemeId))
109 QIcon::setThemeName(themeName);
110 else
111 QIcon::setFallbackThemeName(themeName);
112 QIcon::setFallbackSearchPaths({QStringLiteral(":/icons/%1").arg(themeName)});
113}
